### Safe file count (excludes _old_, hidden dirs, and stray files)

```python
def safe_count_skill_dirs(gh_dir):
    return [d for d in os.listdir(gh_dir)
            if os.path.isdir(os.path.join(gh_dir, d))
            and not d.startswith('.')
            and '_old_' not in d
            and not d.endswith('.md')
            and not d.endswith('.txt')]
```

### Dynamic subdir enumeration (replaces hardcoded subdirs list)

```python
def get_all_subdirs(local_dirs):
    """Enumerate all 1-level subdirs from all profile roots."""
    subs = ['']
    for root in local_dirs:
        if not os.path.isdir(root):
            continue
        for entry in os.listdir(root):
            full = os.path.join(root, entry)
            if os.path.isdir(full):
                subs.append(entry)
    return list(set(subs))
```

### Find local skill dir (returns DIR, not SKILL.md)

```python
def find_local_skill_dir(name, local_dirs, subdirs):
    """Find the skill DIR (not SKILL.md), prefer latest mtime."""
    candidates = []
    for root in local_dirs:
        for sub in subdirs:
            p = os.path.join(root, sub, name) if sub else os.path.join(root, name)
            if os.path.isdir(p):
                real = os.path.realpath(p)
                sm = os.path.join(real, 'SKILL.md')
                if os.path.isfile(sm):
                    candidates.append((p, os.path.getmtime(sm)))
    if not candidates:
        return None
    candidates.sort(key=lambda x: x[1], reverse=True)
    return candidates[0][0]  # the dir path itself
```

### Copy with symlink translation + 精简版 protection

```python
def copy_skill_safely(src_dir, dst_dir):
    """Copy only if local is not a精简版. Otherwise only overwrite SKILL.md."""
    src_real = os.path.realpath(src_dir)

    # Detect精简版: if local has <50% of tracked files, don't wholesale copy
    # (caller should check this first; this is just the safe-copy path)
    if os.path.exists(dst_dir):
        safe_rmtree(dst_dir)
    os.makedirs(dst_dir, exist_ok=True)
    for root, dirs, files in os.walk(src_real, followlinks=True):
        if '__pycache__' in root:
            continue
        rel = os.path.relpath(root, src_real)
        target_root = os.path.join(dst_dir, rel) if rel != '.' else dst_dir
        os.makedirs(target_root, exist_ok=True)
        for f in files:
            if '__pycache__' in f:
                continue
            try:
                shutil.copy2(os.path.join(root, f),
                           os.path.join(target_root, f),
                           follow_symlinks=True)
            except (PermissionError, OSError):
                pass  # skip locked files
```

## Common error transcripts

### `shutil.rmtree` PermissionError on Windows

```
PermissionError: [WinError 5]: 拒绝访问。
'C:\\tmp\\awesome-skills-clean\\answer\\.git\\objects\\00\\42f9872b3a38d4264bc2a0cf89c3001a005de6'
```

**Cause**: Git holds open file handles on `.git/objects/`.
**Fix**: Use `safe_rmtree` (rename-fallback). Cleanup `_old_<ts>` dirs after.

### `cp -rL` nested copy

```bash
$ ls apple-design/
apple-design  references  SKILL.md     # ← nested!
$ ls apple-design/apple-design/
references  SKILL.md
```

**Cause**: dst was an existing directory.
**Fix**: `rm -rf dst` BEFORE copy.

### `git status` shows unexpected `D` for tracked files

```
D wechat-article-archive/scripts/archive_common.py
D wechat-article-archive/scripts/collect_articles.py
...
```

**Cause**: Local install is精简版 (only SKILL.md). Copy deleted tracked files.
**Fix**: `git checkout HEAD -- <skill>/` to restore, then only overwrite SKILL.md.
